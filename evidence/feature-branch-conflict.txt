def sortArray(numbers):
    return sorted(numbers)


def isPrime(n):
    if n <= 1:
        return False
    for i in range(2, n):
        if n % i == 0:
            return False
    return True


def countPrimes(numbers):
    return sum(1 for number in numbers if isPrime(number))
